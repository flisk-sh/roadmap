{%- comment -%}Sets cur_view, cur_subs (the list the sub-tabs come from) and cur_sub_name for the page being built.{%- endcomment -%} {%- assign cur_view = nil -%} {%- assign cur_sub_name = nil -%} {%- assign cur_subs = "" | split: "" -%} {%- if page.view -%} {%- assign cur_view = site.data.meta.views | where: "id", page.view | first -%} {%- case cur_view.subtabs -%} {%- when "families" -%}{%- assign cur_subs = site.data.families -%} {%- when "tracks" -%}{%- assign cur_subs = site.data.journeys.tracks -%} {%- when "roles" -%}{%- assign cur_subs = site.data.roles.groups -%} {%- when "skills" -%}{%- assign cur_subs = site.data.skill_categories -%} {%- endcase -%} {%- if page.sub -%} {%- assign cur_sub_obj = cur_subs | where: "id", page.sub | first -%} {%- assign cur_sub_name = cur_sub_obj.name -%} {%- endif -%} {%- endif -%}
